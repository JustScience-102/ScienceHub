## Phần 1: Cơ sở dữ liệu (`supabase/`)

### 1.1. File mới: `supabase/06_v2_3_audit_log.sql` (Lát cắt V2-3)

File này tạo bảng `audit_log`, khóa quyền RLS chỉ cho Quản lý đọc, chặn sửa/xóa trực tiếp, gắn trigger tự động ghi nhật ký trên các bảng nghiệp vụ, hỗ trợ chu kỳ **Năm học (15/06 năm nay -> 12/06 năm sau)** và hàm làm mới nhật ký đầu năm học mới. Chạy lại được nhiều lần an toàn (Idempotent).

```sql
-- =====================================================================
-- Trung tâm ScienceHub - FILE 06: V2-3 NHẬT KÝ THAY ĐỔI (AUDIT LOG)
-- Chạy SAU 05_v2_2_teacher_access.sql. An toàn khi chạy lại nhiều lần.
-- KHÔNG chạy lại 03_rpc_and_auth.sql.
--
-- Nội dung:
--   1. Hàm tiện ích tính khoảng Năm học của Trung tâm (15/06 -> 12/06 năm sau).
--   2. Bảng public.audit_log + Index + RLS (chỉ Quản lý đọc) + Trigger chặn sửa/xóa.
--   3. Hàm nội bộ _write_audit_log & 6 Trigger tự động bắt thay đổi:
--      - Điểm danh & Học bù (attendance_records, giữ nguyên price_snapshot đóng băng)
--      - Ghi danh, Cho nghỉ, Đổi lớp (class_enrollments)
--      - Thêm/Sửa/Ẩn/Khôi phục Lớp học (classes), Học sinh (students), Giáo viên (teachers)
--      - Liên kết / Khóa / Mở khóa tài khoản Giáo viên (profiles)
--   4. RPC cho Quản lý: query_audit_logs, clear_audit_logs_for_new_school_year.
-- =====================================================================

-- ---------------------------------------------------------------------
-- 1. Hàm tiện ích xác định Năm học (15/06/YYYY -> 12/06/(YYYY+1))
-- ---------------------------------------------------------------------
create or replace function public.vn_school_year_bounds(p_start_year integer default null)
returns jsonb
language plpgsql
stable
security definer
set search_path = public
as $$
declare
  v_today      date := public.vn_today();
  v_year       integer := extract(year from v_today)::integer;
  v_start_year integer;
  v_start_date date;
  v_end_date   date;
begin
  if p_start_year is not null then
    v_start_year := p_start_year;
  else
    if v_today >= make_date(v_year, 6, 15) then
      v_start_year := v_year;
    else
      v_start_year := v_year - 1;
    end if;
  end if;

  v_start_date := make_date(v_start_year, 6, 15);
  v_end_date   := make_date(v_start_year + 1, 6, 12);

  return jsonb_build_object(
    'school_year', v_start_year::text || '-' || (v_start_year + 1)::text,
    'start_year',  v_start_year,
    'end_year',    v_start_year + 1,
    'start_date',  v_start_date,
    'end_date',    v_end_date
  );
end;
$$;

-- ---------------------------------------------------------------------
-- 2. Bảng audit_log & Chốt chặn toàn vẹn (RLS + Chặn UPDATE/DELETE)
--    Các cột UUID liên kết không đặt FK RESTRICT để không cản trở thao tác
--    xóa điểm danh (status = null) hoặc thay thế profile giáo viên.
-- ---------------------------------------------------------------------
create table if not exists public.audit_log (
  id               uuid primary key default gen_random_uuid(),
  created_at       timestamptz not null default now(),
  actor_user_id    uuid,
  actor_role       text not null default 'system'
                   check (actor_role in ('manager', 'teacher', 'system')),
  actor_teacher_id uuid,
  actor_name       text not null default 'Hệ thống',
  action           text not null,
  table_name       text not null,
  record_id        uuid,
  class_id         uuid,
  student_id       uuid,
  summary          text not null,
  old_data         jsonb,
  new_data         jsonb
);

create index if not exists idx_audit_log_created_at
  on public.audit_log (created_at desc);
create index if not exists idx_audit_log_action
  on public.audit_log (action);
create index if not exists idx_audit_log_class_id
  on public.audit_log (class_id) where class_id is not null;
create index if not exists idx_audit_log_student_id
  on public.audit_log (student_id) where student_id is not null;
create index if not exists idx_audit_log_actor_teacher
  on public.audit_log (actor_teacher_id) where actor_teacher_id is not null;

alter table public.audit_log enable row level security;

revoke all on table public.audit_log from anon, public;
revoke insert, update, delete, truncate on table public.audit_log from authenticated;
grant select on table public.audit_log to authenticated;

drop policy if exists v2_manager_select_audit_log on public.audit_log;
create policy v2_manager_select_audit_log on public.audit_log
  for select to authenticated
  using ((select public.is_manager()));

-- Trigger chặn tuyệt đối UPDATE và DELETE trực tiếp trên audit_log.
-- Chỉ mở khóa DELETE tạm thời trong giao dịch khi Quản lý gọi hàm làm mới năm học.
create or replace function public._guard_audit_log_immutable()
returns trigger
language plpgsql
as $$
begin
  if tg_op = 'DELETE'
     and coalesce(current_setting('sciencehub.allow_audit_clear', true), '') = 'on' then
    return old;
  end if;
  raise exception 'Trung tâm: Nhật ký thay đổi được bảo vệ toàn vẹn, không được phép chỉnh sửa hoặc xóa trực tiếp.';
end;
$$;

revoke execute on function public._guard_audit_log_immutable() from anon, authenticated, public;

drop trigger if exists trg_audit_log_immutable on public.audit_log;
create trigger trg_audit_log_immutable
  before update or delete on public.audit_log
  for each row execute function public._guard_audit_log_immutable();

-- ---------------------------------------------------------------------
-- 3. Hàm nội bộ ghi nhận Nhật ký (_write_audit_log)
-- ---------------------------------------------------------------------
create or replace function public._write_audit_log(
  p_action     text,
  p_table_name text,
  p_record_id  uuid,
  p_class_id   uuid,
  p_student_id uuid,
  p_summary    text,
  p_old_data   jsonb,
  p_new_data   jsonb
) returns void
language plpgsql
security definer
set search_path = public
as $$
declare
  v_uid        uuid := auth.uid();
  v_role       text;
  v_teacher_id uuid;
  v_actor_name text;
  v_email      text;
  v_tname      text;
begin
  if v_uid is not null then
    select p.role, p.teacher_id
      into v_role, v_teacher_id
    from public.profiles p
    where p.user_id = v_uid;

    select u.email into v_email
    from auth.users u
    where u.id = v_uid;

    if v_role = 'teacher' and v_teacher_id is not null then
      select t.full_name into v_tname
      from public.teachers t
      where t.id = v_teacher_id;
      v_actor_name := coalesce('GV ' || v_tname, v_email, 'Giáo viên');
    elsif v_role = 'manager' then
      v_actor_name := case
        when v_email is not null then 'Quản lý (' || v_email || ')'
        else 'Quản lý'
      end;
    else
      v_role       := 'system';
      v_actor_name := coalesce(v_email, 'Hệ thống');
    end if;
  else
    v_role       := 'system';
    v_actor_name := 'Hệ thống (SQL)';
  end if;

  insert into public.audit_log (
    actor_user_id, actor_role, actor_teacher_id, actor_name,
    action, table_name, record_id, class_id, student_id,
    summary, old_data, new_data
  ) values (
    v_uid, v_role, v_teacher_id, v_actor_name,
    p_action, p_table_name, p_record_id, p_class_id, p_student_id,
    p_summary, p_old_data, p_new_data
  );
end;
$$;

revoke execute on function public._write_audit_log(text, text, uuid, uuid, uuid, text, jsonb, jsonb)
  from anon, authenticated, public;

-- Hàm tiện ích đổi trạng thái điểm danh sang nhãn tiếng Việt
create or replace function public._attendance_status_label(p_status text, p_price integer)
returns text
language sql
immutable
as $$
  select case p_status
    when 'present' then 'Có mặt (' || coalesce(trim(to_char(p_price, '999G999G999')) || 'đ', '0đ') || ')'
    when 'absent' then 'Vắng tính phí (' || coalesce(trim(to_char(p_price, '999G999G999')) || 'đ', '0đ') || ')'
    when 'absent_excused' then 'Vắng có phép (0đ)'
    else 'Chưa chấm'
  end;
$$;

revoke execute on function public._attendance_status_label(text, integer)
  from anon, authenticated, public;

-- ---------------------------------------------------------------------
-- 4. Các Trigger tự động ghi Nhật ký trên các bảng nghiệp vụ
-- ---------------------------------------------------------------------

-- 4.1 Trigger Điểm danh & Học bù (attendance_records)
--     Giữ nguyên price_snapshot đã đóng băng, không tính lại theo classes.price_per_session.
create or replace function public._trg_audit_attendance()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
declare
  v_rec            public.attendance_records%rowtype;
  v_enr            public.class_enrollments%rowtype;
  v_class_name     text;
  v_orig_class     text;
  v_student_name   text;
  v_student_code   text;
  v_is_makeup      boolean := false;
  v_action         text;
  v_summary        text;
  v_old_json       jsonb := null;
  v_new_json       jsonb := null;
begin
  v_rec := case when tg_op = 'DELETE' then old else new end;

  -- Bỏ qua nếu UPDATE nhưng trạng thái và đơn giá đóng băng không đổi
  if tg_op = 'UPDATE'
     and old.status is not distinct from new.status
     and old.price_snapshot is not distinct from new.price_snapshot
     and old.enrollment_id is not distinct from new.enrollment_id then
    return new;
  end if;

  select * into v_enr from public.class_enrollments where id = v_rec.enrollment_id;
  select c.name into v_class_name from public.classes c where c.id = v_rec.class_id;
  select s.full_name, s.student_code
    into v_student_name, v_student_code
  from public.students s
  where s.id = v_rec.student_id;

  if v_enr.id is not null and v_enr.class_id <> v_rec.class_id then
    v_is_makeup := true;
    select c.name into v_orig_class from public.classes c where c.id = v_enr.class_id;
  end if;

  if tg_op in ('UPDATE', 'DELETE') then
    v_old_json := jsonb_build_object(
      'attendance_date', old.attendance_date,
      'status',          old.status,
      'price_snapshot',  old.price_snapshot,
      'class_id',        old.class_id,
      'class_name',      v_class_name,
      'enrollment_id',   old.enrollment_id,
      'orig_class_id',   v_enr.class_id,
      'orig_class_name', coalesce(v_orig_class, v_class_name),
      'student_id',      old.student_id,
      'student_code',    v_student_code,
      'student_name',    v_student_name,
      'is_makeup',       v_is_makeup
    );
  end if;

  if tg_op in ('INSERT', 'UPDATE') then
    v_new_json := jsonb_build_object(
      'attendance_date', new.attendance_date,
      'status',          new.status,
      'price_snapshot',  new.price_snapshot,
      'class_id',        new.class_id,
      'class_name',      v_class_name,
      'enrollment_id',   new.enrollment_id,
      'orig_class_id',   v_enr.class_id,
      'orig_class_name', coalesce(v_orig_class, v_class_name),
      'student_id',      new.student_id,
      'student_code',    v_student_code,
      'student_name',    v_student_name,
      'is_makeup',       v_is_makeup
    );
  end if;

  if tg_op = 'INSERT' then
    if v_is_makeup then
      v_action  := 'makeup_record';
      v_summary := format(
        'Ghi nhận học bù cho %s (%s) tại lớp %s ngày %s (bù cho lớp %s, đơn giá gốc %s)',
        coalesce(v_student_name, 'Học sinh'),
        coalesce(v_student_code, ''),
        coalesce(v_class_name, ''),
        to_char(new.attendance_date, 'DD/MM/YYYY'),
        coalesce(v_orig_class, ''),
        public._attendance_status_label(new.status, new.price_snapshot)
      );
    else
      v_action  := 'attendance_create';
      v_summary := format(
        'Điểm danh %s (%s) lớp %s ngày %s: %s',
        coalesce(v_student_name, 'Học sinh'),
        coalesce(v_student_code, ''),
        coalesce(v_class_name, ''),
        to_char(new.attendance_date, 'DD/MM/YYYY'),
        public._attendance_status_label(new.status, new.price_snapshot)
      );
    end if;
  elsif tg_op = 'UPDATE' then
    v_action  := 'attendance_update';
    v_summary := format(
      'Đổi điểm danh %s (%s) lớp %s ngày %s: %s -> %s',
      coalesce(v_student_name, 'Học sinh'),
      coalesce(v_student_code, ''),
      coalesce(v_class_name, ''),
      to_char(new.attendance_date, 'DD/MM/YYYY'),
      public._attendance_status_label(old.status, old.price_snapshot),
      public._attendance_status_label(new.status, new.price_snapshot)
    );
  else
    v_action  := 'attendance_delete';
    v_summary := format(
      'Xóa điểm danh%s của %s (%s) tại lớp %s ngày %s (trước đó: %s)',
      case when v_is_makeup then ' học bù' else '' end,
      coalesce(v_student_name, 'Học sinh'),
      coalesce(v_student_code, ''),
      coalesce(v_class_name, ''),
      to_char(old.attendance_date, 'DD/MM/YYYY'),
      public._attendance_status_label(old.status, old.price_snapshot)
    );
  end if;

  perform public._write_audit_log(
    v_action,
    'attendance_records',
    v_rec.id,
    v_rec.class_id,
    v_rec.student_id,
    v_summary,
    v_old_json,
    v_new_json
  );

  return case when tg_op = 'DELETE' then old else new end;
end;
$$;

revoke execute on function public._trg_audit_attendance() from anon, authenticated, public;

drop trigger if exists trg_audit_attendance on public.attendance_records;
create trigger trg_audit_attendance
  after insert or update or delete on public.attendance_records
  for each row execute function public._trg_audit_attendance();

-- 4.2 Trigger Ghi danh / Cho nghỉ / Đổi lớp (class_enrollments)
create or replace function public._trg_audit_enrollments()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
declare
  v_class_name     text;
  v_student_name   text;
  v_student_code   text;
  v_transfer_enr   public.class_enrollments%rowtype;
  v_new_class_name text;
  v_action         text;
  v_summary        text;
begin
  select c.name into v_class_name from public.classes c where c.id = new.class_id;
  select s.full_name, s.student_code
    into v_student_name, v_student_code
  from public.students s
  where s.id = new.student_id;

  if tg_op = 'INSERT' then
    v_action  := 'enrollment_create';
    v_summary := format(
      'Ghi danh %s (%s) vào lớp %s từ ngày %s',
      coalesce(v_student_name, 'Học sinh'),
      coalesce(v_student_code, ''),
      coalesce(v_class_name, ''),
      to_char(new.joined_at, 'DD/MM/YYYY')
    );

    perform public._write_audit_log(
      v_action,
      'class_enrollments',
      new.id,
      new.class_id,
      new.student_id,
      v_summary,
      null,
      jsonb_build_object(
        'class_id',     new.class_id,
        'class_name',   v_class_name,
        'student_id',   new.student_id,
        'student_code', v_student_code,
        'student_name', v_student_name,
        'status',       new.status,
        'joined_at',    new.joined_at
      )
    );
  elsif tg_op = 'UPDATE' and old.status = 'active' and new.status = 'dropped' then
    -- Kiểm tra xem việc nghỉ lớp này có nằm trong giao dịch Đổi lớp (transfer_enrollment) không
    select * into v_transfer_enr
    from public.class_enrollments e
    where e.student_id = new.student_id
      and e.id <> new.id
      and e.status = 'active'
      and e.joined_at = new.left_at
      and e.created_at = transaction_timestamp()
    order by e.created_at desc
    limit 1;

    if found then
      select c.name into v_new_class_name from public.classes c where c.id = v_transfer_enr.class_id;
      v_action  := 'enrollment_transfer';
      v_summary := format(
        'Đổi lớp cho %s (%s): từ lớp %s sang lớp %s (ngày chuyển %s)',
        coalesce(v_student_name, 'Học sinh'),
        coalesce(v_student_code, ''),
        coalesce(v_class_name, ''),
        coalesce(v_new_class_name, ''),
        to_char(new.left_at, 'DD/MM/YYYY')
      );

      perform public._write_audit_log(
        v_action,
        'class_enrollments',
        new.id,
        new.class_id,
        new.student_id,
        v_summary,
        jsonb_build_object(
          'from_class_id',   new.class_id,
          'from_class_name', v_class_name,
          'joined_at',       old.joined_at,
          'status',          old.status
        ),
        jsonb_build_object(
          'to_class_id',     v_transfer_enr.class_id,
          'to_class_name',   v_new_class_name,
          'transfer_date',   new.left_at,
          'new_enrollment',  v_transfer_enr.id,
          'status',          new.status
        )
      );
    else
      v_action  := 'enrollment_drop';
      v_summary := format(
        'Cho %s (%s) nghỉ lớp %s từ ngày %s',
        coalesce(v_student_name, 'Học sinh'),
        coalesce(v_student_code, ''),
        coalesce(v_class_name, ''),
        to_char(new.left_at, 'DD/MM/YYYY')
      );

      perform public._write_audit_log(
        v_action,
        'class_enrollments',
        new.id,
        new.class_id,
        new.student_id,
        v_summary,
        jsonb_build_object(
          'class_id',     new.class_id,
          'class_name',   v_class_name,
          'student_id',   new.student_id,
          'student_code', v_student_code,
          'student_name', v_student_name,
          'status',       old.status,
          'joined_at',    old.joined_at,
          'left_at',      old.left_at
        ),
        jsonb_build_object(
          'class_id',     new.class_id,
          'class_name',   v_class_name,
          'student_id',   new.student_id,
          'student_code', v_student_code,
          'student_name', v_student_name,
          'status',       new.status,
          'joined_at',    new.joined_at,
          'left_at',      new.left_at
        )
      );
    end if;
  end if;

  return new;
end;
$$;

revoke execute on function public._trg_audit_enrollments() from anon, authenticated, public;

drop trigger if exists trg_audit_enrollments on public.class_enrollments;
create trigger trg_audit_enrollments
  after insert or update on public.class_enrollments
  for each row execute function public._trg_audit_enrollments();

-- 4.3 Trigger Lớp học (classes): Thêm / Sửa / Ẩn / Khôi phục
create or replace function public._trg_audit_classes()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
declare
  v_old_teacher text;
  v_new_teacher text;
  v_action      text;
  v_summary     text;
begin
  select t.full_name into v_new_teacher from public.teachers t where t.id = new.teacher_id;

  if tg_op = 'INSERT' then
    v_action  := 'class_create';
    v_summary := format(
      'Tạo lớp %s (Khối %s, GV: %s, đơn giá: %sđ/buổi)',
      new.name,
      coalesce(new.grade::text, '-'),
      coalesce(v_new_teacher, '-'),
      trim(to_char(new.price_per_session, '999G999G999'))
    );

    perform public._write_audit_log(
      v_action,
      'classes',
      new.id,
      new.id,
      null,
      v_summary,
      null,
      jsonb_build_object(
        'name',              new.name,
        'subject',           new.subject,
        'grade',             new.grade,
        'class_no',          new.class_no,
        'teacher_id',        new.teacher_id,
        'teacher_name',      v_new_teacher,
        'price_per_session', new.price_per_session
      )
    );
  elsif tg_op = 'UPDATE' then
    select t.full_name into v_old_teacher from public.teachers t where t.id = old.teacher_id;

    if old.deleted_at is null and new.deleted_at is not null then
      v_action  := 'class_hide';
      v_summary := format('Ẩn lớp %s', new.name);
    elsif old.deleted_at is not null and new.deleted_at is null then
      v_action  := 'class_restore';
      v_summary := format('Khôi phục lớp %s', new.name);
    elsif old.name is distinct from new.name
       or old.subject is distinct from new.subject
       or old.grade is distinct from new.grade
       or old.class_no is distinct from new.class_no
       or old.teacher_id is distinct from new.teacher_id
       or old.price_per_session is distinct from new.price_per_session then
      v_action  := 'class_update';
      v_summary := format(

    end if;

    perform public._write_audit_log(

      'classes',
      new.id,
      new.id,
      null,
      v_summary,
      jsonb_build_object(
        'name',              old.name,
        'subject',           old.subject,
        'grade',             old.grade,
        'class_no',          old.class_no,
        'teacher_id',        old.teacher_id,
        'teacher_name',      v_old_teacher,

        'grade',             new.grade,
        'class_no',          new.class_no,
        'teacher_id',        new.teacher_id,
        'teacher_name',      v_new_teacher,

-- 5. Các RPC dành cho Quản lý tra cứu & làm mới Nhật ký theo Năm học
-- ---------------------------------------------------------------------

-- 5.1 Tra cứu nhật ký thay đổi (có lọc theo khoảng ngày, vai trò, giáo viên, nhóm thao tác, lớp, học sinh, từ khóa)
create or replace function public.query_audit_logs(
  p_from_date      date default null,
  p_to_date        date default null,
  p_actor_role     text default null,
  p_teacher_id     uuid default null,
  p_action_group   text default null,
  p_class_id       uuid default null,
  p_student_id     uuid default null,
  p_search         text default null,
  p_limit          integer default 100,
  p_offset         integer default 0
) returns jsonb
language plpgsql
stable
security definer
set search_path = public
as $$
declare
  v_limit   integer := greatest(1, least(coalesce(p_limit, 100), 500));
  v_offset  integer := greatest(0, coalesce(p_offset, 0));
  v_search  text    := nullif(trim(coalesce(p_search, '')), '');
  v_total   integer;
  v_items   jsonb;
begin
  perform public.assert_manager();

  with filtered as (
    select l.*
    from public.audit_log l
    where (p_from_date is null or (l.created_at at time zone 'Asia/Ho_Chi_Minh')::date >= p_from_date)
      and (p_to_date   is null or (l.created_at at time zone 'Asia/Ho_Chi_Minh')::date <= p_to_date)
      and (p_actor_role is null or p_actor_role = '' or l.actor_role = p_actor_role)
      and (p_teacher_id is null or l.actor_teacher_id = p_teacher_id)
      and (p_class_id   is null or l.class_id = p_class_id)
      and (p_student_id is null or l.student_id = p_student_id)
      and (
        p_action_group is null or p_action_group = ''
        or (p_action_group = 'attendance'  and l.action in ('attendance_create', 'attendance_update', 'attendance_delete'))
        or (p_action_group = 'makeup'      and l.action = 'makeup_record')
        or (p_action_group = 'enrollment'  and l.action in ('enrollment_create', 'enrollment_drop', 'enrollment_transfer'))
        or (p_action_group = 'class'       and l.action in ('class_create', 'class_update', 'class_hide', 'class_restore'))
        or (p_action_group = 'student'     and l.action in ('student_create', 'student_update', 'student_hide', 'student_restore'))
        or (p_action_group = 'teacher'     and l.action in ('teacher_create', 'teacher_update', 'teacher_hide', 'teacher_restore', 'teacher_account_link', 'teacher_account_lock', 'teacher_account_unlock'))
        or l.action = p_action_group
      )
      and (
        v_search is null
        or l.summary ilike '%' || v_search || '%'
        or l.actor_name ilike '%' || v_search || '%'
      )
  )
  select
    count(*),
    coalesce(
      (
        select jsonb_agg(to_jsonb(p.*) order by p.created_at desc)
        from (
          select *
          from filtered
          order by created_at desc
          limit v_limit offset v_offset
        ) p
      ),
      '[]'::jsonb
    )
  into v_total, v_items
  from filtered;

  return jsonb_build_object(
    'total',       coalesce(v_total, 0),
    'limit',       v_limit,
    'offset',      v_offset,
    'school_year', public.vn_school_year_bounds(),
    'items',       v_items
  );
end;
$$;

-- 5.2 Xóa nhật ký cũ sau khi đã tải file sao lưu để bắt đầu Năm học mới (15/06 -> 12/06)
create or replace function public.clear_audit_logs_for_new_school_year(
  p_confirm_text text,
  p_before_date  date default null
) returns integer
language plpgsql
security definer
set search_path = public
as $$
declare
  v_deleted integer := 0;
begin
  perform public.assert_manager();

  if trim(coalesce(p_confirm_text, '')) <> 'XOA NHAT KY' then
    raise exception 'Trung tâm: Vui lòng nhập đúng cụm từ xác nhận "XOA NHAT KY" sau khi đã tải file sao lưu.';
  end if;

  -- Mở khóa tạm thời trong giao dịch hiện tại cho phép xóa dòng trong audit_log
  perform set_config('sciencehub.allow_audit_clear', 'on', true);

  if p_before_date is null then
    delete from public.audit_log;
  else
    delete from public.audit_log
    where (created_at at time zone 'Asia/Ho_Chi_Minh')::date <= p_before_date;
  end if;

  get diagnostics v_deleted = row_count;

  perform set_config('sciencehub.allow_audit_clear', 'off', true);

  -- Ghi lại 1 dòng nhật ký mở đầu năm học mới
  perform public._write_audit_log(
    'audit_log_clear',
    'audit_log',
    null,
    null,
    null,
    format(
      'Quản lý đã sao lưu và làm mới %s dòng nhật ký để bắt đầu năm học mới%s.',
      v_deleted,
      case
        when p_before_date is not null then ' (xóa dữ liệu đến hết ngày ' || to_char(p_before_date, 'DD/MM/YYYY') || ')'
        else ''
      end
    ),
    jsonb_build_object('deleted_count', v_deleted, 'before_date', p_before_date),
    jsonb_build_object('reset_at', now())
  );

  return v_deleted;
end;
$$;
```

### 1.2. File mới: `supabase/07_v2_4_backup.sql` (Lát cắt V2-4)

File này bổ sung 2 hàm RPC dành riêng cho Quản lý (`assert_manager()`) để xuất trọn vẹn bản sao lưu JSON toàn Trung tâm (hoặc riêng Nhật ký năm học) và xuất báo cáo Điểm danh – Học phí theo tháng (cộng từ `price_snapshot` đã đóng băng).

```sql
-- =====================================================================
-- Trung tâm ScienceHub - FILE 07: V2-4 SAO LƯU DỮ LIỆU THỦ CÔNG (JSON/CSV)
-- Chạy SAU 06_v2_3_audit_log.sql. An toàn khi chạy lại nhiều lần.
--
-- Nội dung:
--   1. export_center_backup(p_include_audit_log): Xuất toàn bộ dữ liệu
--      các bảng nghiệp vụ trong 1 giao dịch đồng nhất (chỉ Quản lý).
--   2. export_monthly_attendance_report(p_year, p_month): Xuất bảng tổng hợp
--      điểm danh & học phí tháng theo đúng price_snapshot đóng băng.
-- =====================================================================

-- ---------------------------------------------------------------------
-- 1. Xuất toàn bộ dữ liệu Trung tâm ra JSON (chỉ Quản lý)
-- ---------------------------------------------------------------------
create or replace function public.export_center_backup(
  p_include_audit_log boolean default true
) returns jsonb
language plpgsql
stable
security definer
set search_path = public
as $$
declare
  v_parents     jsonb;
  v_teachers    jsonb;
  v_students    jsonb;
  v_classes     jsonb;
  v_schedules   jsonb;
  v_enrollments jsonb;
  v_attendance  jsonb;
  v_profiles    jsonb;
  v_audit_logs  jsonb := '[]'::jsonb;
begin
  perform public.assert_manager();

  select coalesce(jsonb_agg(to_jsonb(p.*) order by p.parent_code), '[]'::jsonb)
    into v_parents
  from public.parents p;

  select coalesce(jsonb_agg(to_jsonb(t.*) order by t.full_name), '[]'::jsonb)
    into v_teachers
  from public.teachers t;

  select coalesce(jsonb_agg(to_jsonb(s.*) order by s.student_code), '[]'::jsonb)
    into v_students
  from public.students s;

  select coalesce(jsonb_agg(to_jsonb(c.*) order by c.name), '[]'::jsonb)
    into v_classes
  from public.classes c;

  select coalesce(jsonb_agg(to_jsonb(cs.*) order by cs.class_id, cs.day_of_week), '[]'::jsonb)
    into v_schedules
  from public.class_schedules cs;

  select coalesce(jsonb_agg(to_jsonb(ce.*) order by ce.joined_at desc, ce.created_at desc), '[]'::jsonb)
    into v_enrollments
  from public.class_enrollments ce;

  select coalesce(jsonb_agg(to_jsonb(ar.*) order by ar.attendance_date desc, ar.created_at desc), '[]'::jsonb)
    into v_attendance
  from public.attendance_records ar;

  select coalesce(
    jsonb_agg(
      jsonb_build_object(
        'user_id',      pr.user_id,
        'email',        u.email,
        'role',         pr.role,
        'teacher_id',   pr.teacher_id,
        'active',       pr.active,
        'created_at',   pr.created_at
      )
      order by pr.role, pr.created_at
    ),
    '[]'::jsonb
  )
  into v_profiles
  from public.profiles pr
  left join auth.users u on u.id = pr.user_id;

  if coalesce(p_include_audit_log, true) then
    select coalesce(jsonb_agg(to_jsonb(al.*) order by al.created_at desc), '[]'::jsonb)
      into v_audit_logs
    from public.audit_log al;
  end if;

  return jsonb_build_object(
    'metadata', jsonb_build_object(
      'center',            'Trung tâm ScienceHub',
      'version',           'V2-4',
      'exported_at',       now(),
      'vn_date',           public.vn_today(),
      'school_year',       public.vn_school_year_bounds(),
      'include_audit_log', coalesce(p_include_audit_log, true),
      'counts', jsonb_build_object(
        'parents',            jsonb_array_length(v_parents),
        'teachers',           jsonb_array_length(v_teachers),
        'students',           jsonb_array_length(v_students),
        'classes',            jsonb_array_length(v_classes),
        'class_schedules',    jsonb_array_length(v_schedules),
        'class_enrollments',  jsonb_array_length(v_enrollments),
        'attendance_records', jsonb_array_length(v_attendance),
        'profiles',           jsonb_array_length(v_profiles),
        'audit_log',          jsonb_array_length(v_audit_logs)
      )
    ),
    'tables', jsonb_build_object(
      'parents',            v_parents,
      'teachers',           v_teachers,
      'students',           v_students,
      'classes',            v_classes,
      'class_schedules',    v_schedules,
      'class_enrollments',  v_enrollments,
      'attendance_records', v_attendance,
      'profiles',           v_profiles,
      'audit_log',          v_audit_logs
    )
  );
end;
$$;

-- ---------------------------------------------------------------------
-- 2. Xuất báo cáo Điểm danh & Học phí theo tháng (chỉ Quản lý)
--    Tính tiền tuyệt đối từ attendance_records.price_snapshot (đơn giá đóng băng).
-- ---------------------------------------------------------------------
create or replace function public.export_monthly_attendance_report(
  p_year  integer,
  p_month integer
) returns jsonb
language plpgsql
stable
security definer
set search_path = public
as $$
declare
  v_start date;
  v_next  date;
  v_rows  jsonb;
begin
  perform public.assert_manager();

  if p_year is null or p_month is null
     or p_year not between 2000 and 2100 or p_month not between 1 and 12 then
    raise exception 'Trung tâm: Tháng hoặc năm xuất báo cáo không hợp lệ.';
  end if;

  v_start := make_date(p_year, p_month, 1);
  v_next  := (v_start + interval '1 month')::date;

  select coalesce(
    jsonb_agg(
      jsonb_build_object(
        'student_code',     x.student_code,
        'student_name',     x.student_name,
        'student_phone',    x.student_phone,
        'grade',            x.grade,
        'parent_code',      x.parent_code,
        'parent_name',      x.parent_name,
        'parent_phone',     x.parent_phone,
        'class_name',       x.class_name,
        'subject',          x.subject,
        'teacher_name',     x.teacher_name,
        'enrollment_status',x.enrollment_status,
        'present_count',    x.present_count,
        'makeup_count',     x.makeup_count,
        'absent_count',     x.absent_count,
        'excused_count',    x.excused_count,
        'charged_sessions', x.charged_sessions,
        'total_tuition',    x.total_tuition
      )
      order by x.class_name, x.student_name
    ),
    '[]'::jsonb
  )
  into v_rows
  from (
    select
      s.student_code,
      s.full_name as student_name,
      coalesce(s.phone, '') as student_phone,
      coalesce(c.grade, s.grade) as grade,
      coalesce(p.parent_code, '') as parent_code,
      coalesce(p.full_name, '') as parent_name,
      coalesce(p.phone, '') as parent_phone,
      c.name as class_name,
      coalesce(c.subject, '') as subject,
      coalesce(t.full_name, '') as teacher_name,
      e.status as enrollment_status,
      count(a.id) filter (where a.status = 'present' and a.class_id = e.class_id) as present_count,
      count(a.id) filter (where a.status = 'present' and a.class_id <> e.class_id) as makeup_count,
      count(a.id) filter (where a.status = 'absent') as absent_count,
      count(a.id) filter (where a.status = 'absent_excused') as excused_count,
      count(a.id) filter (where a.status in ('present', 'absent')) as charged_sessions,
      coalesce(sum(a.price_snapshot) filter (where a.status in ('present', 'absent')), 0) as total_tuition
    from public.class_enrollments e
    join public.students s on s.id = e.student_id
    left join public.parents p on p.id = s.parent_id
    join public.classes c on c.id = e.class_id
    left join public.teachers t on t.id = c.teacher_id
    left join public.attendance_records a
      on a.enrollment_id = e.id
     and a.attendance_date >= v_start
     and a.attendance_date < v_next
    where e.joined_at < v_next
      and (e.left_at is null or e.left_at >= v_start)
    group by
      s.id, s.student_code, s.full_name, s.phone, s.grade,
      p.parent_code, p.full_name, p.phone,
      c.id, c.name, c.subject, c.grade,
      t.full_name, e.id, e.status
  ) x;

  return jsonb_build_object(
    'year',  p_year,
    'month', p_month,
    'rows',  v_rows
  );
end;
$$;
```

```javascript
export function getVnTodayISO() {
  const parts = new Intl.DateTimeFormat('en-CA', {
    timeZone: 'Asia/Ho_Chi_Minh',
    year: 'numeric',
    month: '2-digit',
    day: '2-digit',
  }).formatToParts(new Date());
  const y = parts.find((p) => p.type === 'year')?.value;
  const m = parts.find((p) => p.type === 'month')?.value;

/**

 */
export function formatVnDateTime(isoTimestamp) {
  if (!isoTimestamp) return '-';
  try {
    const dt = new Date(isoTimestamp);
    return new Intl.DateTimeFormat('vi-VN', {
      timeZone: 'Asia/Ho_Chi_Minh',
      day: '2-digit',
      month: '2-digit',
      year: 'numeric',
      hour: '2-digit',
      minute: '2-digit',
      second: '2-digit',
      hour12: false,
    }).format(dt);
  } catch {
    return String(isoTimestamp);
  }
}

export function formatAuditPayload(data) {
  if (!data || typeof data !== 'object') return [];
  const entries = [];
  for (const [key, rawVal] of Object.entries(data)) {
    if (
      key.endsWith('_id') ||
      key === 'new_enrollment' ||
      rawVal === undefined
    ) {
      continue;
    }
    const label = FIELD_LABELS[key] || key;
    let display = rawVal;

    if (rawVal === null || rawVal === '') {
      display = 'Trống';
    } else if (key === 'price_snapshot' || key === 'price_per_session') {
      display = formatCurrencyVnd(rawVal);
    } else if (key === 'status' && STATUS_VALUE_LABELS[rawVal]) {
      display = STATUS_VALUE_LABELS[rawVal];
    } else if (typeof rawVal === 'boolean') {
      display = rawVal ? 'Có' : 'Không';
    } else if (
      (key.endsWith('_date') || key === 'joined_at' || key === 'left_at') &&
      typeof rawVal === 'string' &&
      /^\d{4}-\d{2}-\d{2}$/.test(rawVal)
    ) {
      const [y, m, d] = rawVal.split('-');
      display = `${d}/${m}/${y}`;
    }

    entries.push({ key, label, value: String(display) });
  }
  return entries;
}
```

```javascript
import { supabase } from './supabaseClient';

/**
 * So sánh tên tiếng Việt theo TÊN GỌI (từ cuối của họ tên) đúng quy ước RULES.md
 */
export function compareVietnameseGivenNames(fullNameA = '', fullNameB = '') {
  const cleanA = String(fullNameA).trim();
  const cleanB = String(fullNameB).trim();
  const partsA = cleanA.split(/\s+/);
  const partsB = cleanB.split(/\s+/);
  const givenA = partsA[partsA.length - 1] || '';
  const givenB = partsB[partsB.length - 1] || '';

  const cmpGiven = givenA.localeCompare(givenB, 'vi', { sensitivity: 'base' });
  if (cmpGiven !== 0) return cmpGiven;
  return cleanA.localeCompare(cleanB, 'vi', { sensitivity: 'base' });
}

function escapeCsvCell(value) {
  if (value === null || value === undefined) return '';
  const str = String(value);
  if (/[",\r\n]/.test(str)) {
    return `"${str.replace(/"/g, '""')}"`;
  }
  return str;
}

export function buildCsvContent(headers, rows) {
  const headerLine = headers.map((h) => escapeCsvCell(h.label)).join(',');
  const bodyLines = rows.map((row) =>
    headers.map((h) => escapeCsvCell(row[h.key])).join(',')
  );
  // Gắn UTF-8 BOM (\uFEFF) để Excel trên Windows/macOS nhận đúng tiếng Việt
  return '\uFEFF' + [headerLine, ...bodyLines].join('\r\n');
}

export function triggerFileDownload(content, filename, mimeType) {
  const blob = new Blob([content], { type: mimeType });
  const url = URL.createObjectURL(blob);
  const link = document.createElement('a');
  link.href = url;
  link.download = filename;
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
  setTimeout(() => URL.revokeObjectURL(url), 1500);
}

function formatDateDdMmYyyy(isoDate) {
  if (!isoDate || typeof isoDate !== 'string') return '';
  const m = isoDate.match(/^(\d{4})-(\d{2})-(\d{2})/);
  if (!m) return isoDate;
  return `${m[3]}/${m[2]}/${m[1]}`;
}

export async function downloadSchoolYearAuditBackup({
  fromDate,
  toDate,
  format = 'json',
} = {}) {
  const bounds = getCurrentSchoolYearBounds();
  const start = fromDate || bounds.startDate;
  const end = toDate || bounds.endDate;

  const { data, error } = await supabase.rpc('query_audit_logs', {
    p_from_date: start,
    p_to_date: end,
    p_limit: 500,
    p_offset: 0,
  });
  if (error) throw error;

  // Nếu có nhiều hơn 500 dòng trong năm học, tải tiếp các trang còn lại
  let allItems = [...(data?.items || [])];
  const total = Number(data?.total || allItems.length);
  let offset = 500;
  while (allItems.length < total && offset < 20000) {
    const { data: nextPage, error: pageErr } = await supabase.rpc(
      'query_audit_logs',
      {
        p_from_date: start,
        p_to_date: end,
        p_limit: 500,
        p_offset: offset,

    );
    if (pageErr) throw pageErr;
    const batch = nextPage?.items || [];
    if (batch.length === 0) break;
    allItems = allItems.concat(batch);
    offset += 500;
  }

  const today = getVnTodayISO();
  const baseName = `sciencehub_nhat_ky_nam_hoc_${bounds.schoolYear}_${today}`;

  if (format === 'csv') {
    const headers = [

export async function downloadClassesCsv() {

    p_include_audit_log: false,

    { key: 'status', label: 'Trạng thái' },

  const rows = (tables.classes || []).map((c) => {
    const teacher = teachersById.get(c.teacher_id);
    const schList = schedulesByClass.get(c.id) || [];
    return {
      name: c.name,
      subject: c.subject ?? '',
      grade: c.grade ?? '',
```

## Phần 4: Script Dọn Sạch `mock_data` Trước Khi Nhập Dữ Liệu Thật (Dành cho V2-4)

Khi bạn đã nghiệm thu xong V2-3 và V2-4 trên dữ liệu thử và chuẩn bị nhập dữ liệu học sinh/phụ huynh thật của Trung tâm, hãy:

1. Vào màn hình **Tổng quan -> Sao lưu & Năm học** bấm tải thử 1 bản `.json` để kiểm tra.
2. Nếu trước đó bạn có chạy `mock_data.sql`, chạy đoạn SQL dưới đây trong **Supabase SQL Editor** để xóa sạch dữ liệu giả lập (giữ nguyên tài khoản Quản lý trong `profiles`) trước khi nhập dữ liệu thật:



```sql
-- CHỈ CHẠY KHI MUỐN XÓA TOÀN BỘ DỮ LIỆU THỬ (MOCK DATA) ĐỂ BẮT ĐẦU VẬN HÀNH THẬT
begin;
  select set_config('sciencehub.allow_audit_clear', 'on', true);
  delete from public.attendance_records;
  delete from public.class_enrollments;
  delete from public.class_schedules;
  delete from public.classes;
  delete from public.students;
  delete from public.parents;
  delete from public.profiles where role = 'teacher';
  delete from public.teachers;
  delete from public.audit_log;
  alter sequence public.parent_code_seq restart with 1;
  alter sequence public.student_code_seq restart with 1;
  select set_config('sciencehub.allow_audit_clear', 'off', true);
commit;

```
